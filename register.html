2. ตัวอย่าง Google Apps Script (Backend Webhook)

                    
คัดลอกโค้ดนี้ลงใน Google Apps Script เพื่อรับค่าจากฟอร์มบันทึกลง Google Sheet และส่ง Push Notification หาพนักงาน:

                    
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  // 1. บันทึกข้อมูลลง Google Sheet
  sheet.appendRow([
    new Date(),
    data.empId,
    data.fullName,
    data.department,
    data.phone,
    data.email,
    data.lineUserId,
    data.linePictureUrl
  ]);
  
  // 2. ส่ง Push Message กลับหา LINE User ID ที่ดักจับมาได้
  var channelAccessToken = "YOUR_LINE_CHANNEL_ACCESS_TOKEN";
  var url = "https://api.line.me/v2/bot/message/push";
  
  var payload = {
    to: data.lineUserId,
    messages: [{
      type: "text",
      text: "🔷 ลงทะเบียนพนักงานสำเร็จ!\nยินดีต้อนรับคุณ " + data.fullName + " (" + data.department + ")"
    }]
  };
  
  UrlFetchApp.fetch(url, {
    method: "post",
    headers: {
      "Content-Type": "application/json",
      "Authorization": "Bearer " + channelAccessToken
    },
    payload: JSON.stringify(payload)
  });
  
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MIME.JSON);
}
